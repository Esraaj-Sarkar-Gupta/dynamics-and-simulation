\section{Triple Pendulum}
In this section, different methods are used to solve for a Triple Pendulum system with discrete point masses $m_1, m_2, m_3$, each connected by rigid connections of length $l_1, l_2, l_3$, making angles $\theta_1, \theta_2, \theta_3$ with the normal. Unless mentioned otherwise, all results presented follow $m_i = l_i = 1.0$, with initial conditions $\theta_i^\text{initial} = \frac{\pi}{2}$ and $\omega_i^\text{initial} = 0.0$; for $i = 1,2,3$.

\subsection{The Lagrangian Formalism}
The system is first solved using the Lagrangian Formalism, for which the net potential energy and net kinetic energy have to be expressed. The potential energy $\mathcal{V}$ is

\begin{equation}
    \mathcal{V} = -m_1gl_1\cos(\theta_1) - m_2g \left\{ l_1\cos(\theta_1) + l_2\cos(\theta_2) \right\} -
    m_3g \left\{l_1\cos(\theta_1) + l_2\cos(\theta_2) + l_3\cos(\theta_3) \right\}
\end{equation}

and the net kinetic energy $\mathcal{T}$ is

\begin{align}
    T &= \frac{1}{2} m_1 \left( v_{x1}^2 + v_{y1}^2 \right) + \frac{1}{2} m_2 \left( v_{x2}^2 + v_{y2}^2 \right) + \frac{1}{2} m_3 \left( v_{x3}^2 + v_{y3}^2 \right)
\end{align}

where

\begin{align}
    v_{x1} &= l_1 \cos(q_1) \, \dot{q}_1 \\
    v_{y1} &= l_1 \sin(q_1) \, \dot{q}_1 \\[0.5em]
    v_{x2} &= v_{x1} + l_2 \cos(q_2) \, \dot{q}_2 \\
    v_{y2} &= v_{y1} + l_2 \sin(q_2) \, \dot{q}_2 \\[0.5em]
    v_{x3} &= v_{x2} + l_3 \cos(q_3) \, \dot{q}_3 \\
    v_{y3} &= v_{y2} + l_3 \sin(q_3) \, \dot{q}_3.
\end{align}

The Lagrangian $\mathcal{L} = \mathcal{T} - \mathcal{V}$ is placed in the Euler-Lagrange Equation Eq. \ref{eq:Euler Lagrange Equation}, the computations for which are done using Julia's \texttt{Symbolics} package. The simplified expressions are neither computed not reported, since the simplification is computationally expensive.

\begin{minted}{julia}
@variables t
d = Differential(t)
dd = Differential(t)^2

@variables m1 m2 m3 l1 l2 l3 g
@variables θ1(t) θ2(t) θ3(t)
@variables q1 q2 q3 dq1 dq2 dq3

# Kinematics and Energy
V = -g * (m1*l1*cos(q1) + m2*l1*cos(q1) + l2*cos(q2)) + m3*(l1*cos(q1) + l2*cos(q2) + l3*cos(q3)))

# Unsigned velocities (using simple chain-rule differentiation)
vx1 = l1 * cos(q1) * dq1
vy1 = l1 * sin(q1) * dq1
vx2 = vx1 + l2 * cos(q2) * dq2
vy2 = vy1 + l2 * sin(q2) * dq2
vx3 = vx2 + l3 * cos(q3) * dq3
vy3 = vy2 + l3 * sin(q3) * dq3

T = 0.5 * m1 * (vx1^2 + vy1^2) + 0.5 * m2 * (vx2^2 + vy2^2) + 0.5 * m3 * (vx3^2 + vy3^2)
L = T - V

# Partials evaluated using static variables
sub_back = Dict(q1=>θ1, q2=>θ2, q3=>θ3, dq1=>d(θ1), dq2=>d(θ2), dq3=>d(θ3))
pL_pθ1 = substitute(expand_derivatives(Differential(q1)(L)), sub_back)
pL_pθ2 = substitute(expand_derivatives(Differential(q2)(L)), sub_back)
pL_pθ3 = substitute(expand_derivatives(Differential(q3)(L)), sub_back)
pL_pdθ1 = substitute(expand_derivatives(Differential(dq1)(L)), sub_back)
pL_pdθ2 = substitute(expand_derivatives(Differential(dq2)(L)), sub_back)
pL_pdθ3 = substitute(expand_derivatives(Differential(dq3)(L)), sub_back)

# Euler-Lagrange equations (LHS = 0)
eq1_lhs = expand_derivatives(d(pL_pdθ1)) - pL_pθ1
eq2_lhs = expand_derivatives(d(pL_pdθ2)) - pL_pθ2
eq3_lhs = expand_derivatives(d(pL_pdθ3)) - pL_pθ3
\end{minted}

The Euler-Lagrange Equations for any system are always linear with respect to the generalized accelerations $\ddot{q}$. The governing equations naturally take the form

\begin{equation}
    \undertilde{E}(\theta, \dot\theta \ddot\theta) = \undertilde{M}(\theta) \ddot{\theta} - \bar{f}(\theta, \dot{theta}) = 0
\end{equation}

where $\undertilde{M}(\theta)$  is the configuration dependent \textit{Mass Matrix} and $\bar{f}$ is the \textit{forcing vector} containing all velocity dependent and position dependent terms.

\begin{enumerate}
    \item \textbf{Extracting the mass matrix: } This is done by finding the Jacobian of $E$ with respect to the second order derivative terms $\ddot\theta$.
        \begin{equation}
            \undertilde{J} = \frac{\partial}{\partial \ddot\theta_j} \undertilde{E}
        \end{equation}
        This eliminates all $\bar{f}$ terms.

        \begin{minted}{julia}
        M_sym = Symbolics.jacobian(eqs_lhs, vars_dd)
        \end{minted}

    \item \textbf{Isolating the forcing vector: } The system is isolated at $\ddot\theta_j = 0 \forall j$, which results in
        \begin{equation}
            \undertilde{E}\Big\vert{}_{\ddot{\theta}=\mathbf{0}} =
            \undertilde{M}(\theta)\cdot \mathbf{0} - \bar{f}(\theta, \dot{\theta}) =
            -\bar{f}(\theta, \dot{\theta})
        \end{equation}

        the negation of which directly yields the forcing vector.

        \begin{minted}{julia}
        zero_dd = Dict(dd(θ1)=>0, dd(θ2)=>0, dd(θ3)=>0)
        f_sym = substitute.(-1 .* eqs_lhs, (zero_dd,))
        \end{minted}
\end{enumerate}

Once the functions have been identified, they must be represented in a way that the Julia package \texttt{DifferentialEquations} is able to interact with them.

\begin{enumerate}
    \item Now, in order to convert the dynamics from time-dependent symbolic expressions to native julia machine code, static algebraic variables must be used.
    
        \begin{minted}{julia}
        @variables th1 th2 th3 w1 w2 w3
        state_subs = Dict(θ1=>th1, θ2=>th2, θ3=>th3, d(θ1)=>w1, d(θ2)=>w2, d(θ3)=>w3)

        M_clean = substitute.(M_sym, (state_subs,))
        f_clean = substitute.(f_sym, (state_subs,))
        \end{minted}

    \item The function arguments must now be structured for \texttt{DifferentialEquations}, and are grouped into parameters and state variables.
    
        \begin{minted}{julia}
        state_vars = [th1, th2, th3, w1, w2, w3]
        param_vars = [m1, m2, m3, l1, l2, l3, g]
        \end{minted}

    \item Now the functions are built using

        \begin{minted}{julia}
        M_func = build_function(M_clean, state_vars, param_vars, expression=Val{false})[1]
        f_func = build_function(f_clean, state_vars, param_vars, expression=Val{false})[1]
        \end{minted}

        where the symbolic Abstract Syntax Tree (AST) of the cleaned expressions is read and compiles optimized Julia code directly into memory. \texttt{expression=Val{false}} tells \texttt{Symbolics} to JIT-compile the generated code into an executable run-time function handle immediately, instead of returning raw Julia code or AST objects.
\end{enumerate}

\subsection{Differential Algebraic Equations Formalism}
Expressing this system in the DAE formalism, firstly three constraints are considered.

\begin{align}
    \Phi_1 &= x_1^2 + y_1^2 - l_1^2 = 0 \\
    \Phi_2 &= (\Delta x_2)^2 + (\Delta y_2)^2 -  l_2^2 = 0
    \Phi_3 &= (\Delta x_3)^2 _ (\Delta y_3)^2 - l_3^2 = 0
\end{align}

where $\Delta X_i = X_i - X_{i-1}$; $X = x, y$ and $i = 2,3$.

The constraint conditions derived from Newton's Second Law of Motion are expressed with the Lagrange Multipliers $\lambda_{1,2,3}$. All tensile constraint forces act proportionate to the direction of the tension:

\begin{equation}
    F_\mathrm{tension} = \mathrm{const} \cdot
    \begin{pmatrix}
        x \\
        y
    \end{pmatrix}
\end{equation}

Thus, the following six equations are produced.

\begin{align}
    m_1 \ddot{x}_1 &= -\lambda_1 x_1 + \lambda_2 \Delta x_2 \\
    m_1 \ddot{y}_1 &= -m_1 g -\lambda_1 y_1 + \lambda_2 \Delta y_2 \\
    m_2 \ddot{x}_2 &= -\lambda_2 \Delta x_2 + \lambda_3 \Delta x_3 \\
    m_2 \ddot{y}_2 &= -m_2 g - \lambda_2 \Delta y_2 + \lambda_3 \Delta y_3 \\
    m_3 \ddot{x}_3 &= -\lambda_3 \Delta x_3\\
    m_3 \ddot{y}_3 &= -m_3 g - \lambda_3 \delta y_3
\end{align}

This is expressed and solved numerically as
\begin{equation}
    \begin{pmatrix}
    m_1 & 0 & 0 & 0 & 0 & 0 & x_1 & -\Delta x_2 & 0 \\
    0 & m_1 & 0 & 0 & 0 & 0 & y_1 & -\Delta y_2 & 0 \\
    0 & 0 & m_2 & 0 & 0 & 0 & 0 & \Delta x_2 & -\Delta x_3 \\
    0 & 0 & 0 & m_2 & 0 & 0 & 0 & \Delta y_2 & -\Delta y_3 \\
    0 & 0 & 0 & 0 & m_3 & 0 & 0 & 0 & \Delta x_3 \\
    0 & 0 & 0 & 0 & 0 & m_3 & 0 & 0 & \Delta y_3 \\
    x_1 & y_1 & 0 & 0 & 0 & 0 & 0 & 0 & 0 \\
    -\Delta x_2 & -\Delta y_2 & \Delta x_2 & \Delta y_2 & 0 & 0 & 0 & 0 & 0 \\
    0 & 0 & -\Delta x_3 & -\Delta y_3 & \Delta x_3 & \Delta y_3 & 0 & 0 & 0
    \end{pmatrix}
    \begin{pmatrix}
    \ddot{x}_1 \\
    \ddot{y}_1 \\
    \ddot{x}_2 \\
    \ddot{y}_2 \\
    \ddot{x}_3 \\
    \ddot{y}_3 \\
    \lambda_1 \\
    \lambda_2 \\
    \lambda_3
    \end{pmatrix}
    =
    \begin{pmatrix}
    0 \\
    -m_1 g \\
    0 \\
    -m_2 g \\
    0 \\
    -m_3 g \\
    -(\dot{x}_1^2 + \dot{y}_1^2) \\
    -(\Delta \dot{x}_2^2 + \Delta \dot{y}_2^2) \\
    -(\Delta \dot{x}_3^2 + \Delta \dot{y}_3^2)
    \end{pmatrix}
\end{equation}

The constraint equations are differeniated twice with respect to time symbolically.

\begin{align}
\frac{d^2\Phi_1}{dt^2} &= 2\left(\dot{x}_1^2 + \dot{y}_1^2\right) + 2 x_1 \ddot{x}_1 + 2 y_1 \ddot{y}_1 = 0 \\[1.5ex]
\frac{d^2\Phi_2}{dt^2} &= 2\left[\left(\dot{x}_2 - \dot{x}_1\right)^2 + \left(\dot{y}_2 - \dot{y}_1\right)^2\right] + 2(x_2 - x_1)(\ddot{x}_2 - \ddot{x}_1) + 2(y_2 - y_1)(\ddot{y}_2 - \ddot{y}_1) = 0 \\[1.5ex]
\frac{d^2\Phi_3}{dt^2} &= 2\left[\left(\dot{x}_3 - \dot{x}_2\right)^2 + \left(\dot{y}_3 - \dot{y}_2\right)^2\right] + 2(x_3 - x_2)(\ddot{x}_3 - \ddot{x}_2) + 2(y_3 - y_2)(\ddot{y}_3 - \ddot{y}_2) = 0
\end{align}

The system of equations (in matrix form) is solved numerically, and results are presented.

\subsection{Newton--Euler Formalism}
\label{sec:triple_newton_euler}

The triple pendulum may also be described using the Newton--Euler formalism,
which derives the equations of motion directly from the forces and moments
acting on the system. As in the preceding formulations, the rods are rigid
and massless, the joints are frictionless, and the masses are concentrated at
their respective joints. The angles $\theta_i$ are measured from the downward
vertical. We take moments about the three joints: the fixed support $O$, the
joint $A$ at $m_1$, and the joint $B$ at $m_2$. The last two joints accelerate,
so their equations must include moving-pivot corrections.

\paragraph{Kinematics.}
Set $\mathbf r_0=\mathbf 0$ and introduce the unit vectors
\begin{equation}
    \mathbf e_i =
    \begin{pmatrix}\sin\theta_i\\-\cos\theta_i\end{pmatrix},
    \qquad
    \mathbf e_i^\perp =
    \begin{pmatrix}\cos\theta_i\\\sin\theta_i\end{pmatrix}.
\end{equation}
The position, velocity, and acceleration of the $k$-th mass are
\begin{align}
    \mathbf r_k &= \sum_{i=1}^{k} l_i\mathbf e_i, \\
    \mathbf v_k &= \sum_{i=1}^{k} l_i\dot\theta_i\mathbf e_i^\perp, \\
    \mathbf a_k &= \sum_{i=1}^{k} l_i\left(
        \ddot\theta_i\mathbf e_i^\perp
        -\dot\theta_i^2\mathbf e_i\right),
    \qquad k=1,2,3.
    \label{eq:triple_ne_kinematics}
\end{align}
The support has $\mathbf v_0=\mathbf a_0=\mathbf 0$. The velocity and
acceleration expressions are evaluated symbolically from the positions in
\texttt{symbolics.jl}, using the chain rule.

\paragraph{Moment equations at the three joints.}
For an inertially fixed point $P$, Euler's angular-momentum equation reads
\begin{equation}
    \sum\boldsymbol\tau_P^{\mathrm{ext}}
    = \frac{d\mathbf H_P}{dt},
    \qquad
    \mathbf H_P=\sum_k(\mathbf r_k-\mathbf r_P)
        \times m_k\mathbf v_k.
\end{equation}
For a moving point $P$ it is useful instead to define angular momentum using
\emph{relative} velocities. The corresponding moment equation for a subchain
of particles is
\begin{equation}
    \boxed{
    \sum\boldsymbol\tau_P^{\mathrm{ext}}
    =\frac{d\mathbf H_{P,\mathrm{rel}}}{dt}
    +\sum_k(\mathbf r_k-\mathbf r_P)\times m_k\mathbf a_P
    }
    \label{eq:triple_ne_moving_pivot}
\end{equation}
where
$\mathbf H_{P,\mathrm{rel}}=\sum_k(\mathbf r_k-\mathbf r_P)
\times m_k(\mathbf v_k-\mathbf v_P)$.
The additional term is essential because $P$ need not be inertially fixed.
It is automatically zero when the pivot is unaccelerated. Since the motion is planar, only
the $z$-components of these vector equations are nonzero.

The three moment equations are applied to progressively shorter subchains:
about $O$ we consider all three masses; about $A$ we consider masses $m_2$
and $m_3$; and about $B$ we consider only $m_3$. This choice eliminates the
unknown joint reactions from each moment equation. In particular, the force
transmitted at the proximal joint acts along its connecting rod and produces
no moment about that joint, while the internal forces of the subchain cancel.

To express all three equations uniformly, let $P_{j-1}$ denote the joint at
$\mathbf r_{j-1}$, with $P_0=O$, $P_1=A$, and $P_2=B$. For $j=1,2,3$ define
\begin{align}
    H_j &:= \sum_{k=j}^{3}
        \left[(\mathbf r_k-\mathbf r_{j-1})\times
        m_k(\mathbf v_k-\mathbf v_{j-1})\right]_z,
        \label{eq:triple_ne_Hj}\\
    C_j &:= \sum_{k=j}^{3}
        \left[(\mathbf r_k-\mathbf r_{j-1})\times
        m_k\mathbf a_{j-1}\right]_z,
        \label{eq:triple_ne_Cj}\\
    \tau_j^{(g)} &:= -g\sum_{k=j}^{3}
        m_k(x_k-x_{j-1}).
        \label{eq:triple_ne_tau}
\end{align}
The residuals of the three Newton--Euler equations are then
\begin{equation}
    E_j:=\frac{dH_j}{dt}+C_j-\tau_j^{(g)}=0,
    \qquad j=1,2,3.
    \label{eq:triple_ne_raw}
\end{equation}
Thus $C_1=0$, whereas $C_2$ and $C_3$ account for the accelerations of
joints $A$ and $B$, respectively. Although the moment equations can be
expanded by hand, doing so yields lengthy expressions involving several
angular-acceleration terms and trigonometric couplings. Here they are formed
and differentiated in Julia using \texttt{Symbolics.jl}, rather than being
introduced as precomputed equations.

\paragraph{Symbolic extraction of the equations of motion.}
The three residuals in Eq.~\eqref{eq:triple_ne_raw} are not yet the most
convenient equations to integrate. Subtracting successive joint-moment
equations isolates the generalized equations associated with each angle:
\begin{equation}
    \mathbf E =
    \begin{pmatrix}
        E_1-E_2\\E_2-E_3\\E_3
    \end{pmatrix}=\mathbf 0.
    \label{eq:triple_ne_reduced}
\end{equation}
Since these equations are linear in the angular accelerations, they have the
form
\begin{equation}
    \mathbf E(\boldsymbol\theta,\dot{\boldsymbol\theta},
               \ddot{\boldsymbol\theta})
    =\mathbf M(\boldsymbol\theta)\ddot{\boldsymbol\theta}
    -\mathbf b(\boldsymbol\theta,\dot{\boldsymbol\theta})=\mathbf0.
    \label{eq:triple_ne_affine}
\end{equation}
As in the Lagrangian implementation, the coefficient matrix is obtained from
the Jacobian of the residuals with respect to angular acceleration, and the
forcing vector from their value at zero angular acceleration:
\begin{equation}
    M_{ij}=\frac{\partial E_i}{\partial\ddot\theta_j},
    \qquad
    \mathbf b=-\left.\mathbf E\right|_{\ddot{\boldsymbol\theta}=\mathbf0}.
    \label{eq:triple_ne_extract}
\end{equation}
The essential steps in \texttt{symbolics.jl} are
\begin{minted}{julia}
# residuals[j] = dH_j/dt + C_j - tau_gravity[j]
equations = [residuals[1] - residuals[2],
             residuals[2] - residuals[3], residuals[3]]
M_sym = Symbolics.jacobian(equations, α)
zero_alpha = Dict(α1=>0, α2=>0, α3=>0)
b_sym = [-substitute(E, zero_alpha) for E in equations]
\end{minted}
The actual file constructs $E_1,E_2,E_3$ from the position vectors, relative
angular momenta, gravitational moments, and moving-pivot corrections before
performing this extraction. To avoid unnecessarily expanding the time
derivative of every angular-momentum product, it uses the equivalent identity
$\frac{d}{dt}(\mathbf r_{k/P}\times m_k\mathbf v_{k/P})
=\mathbf r_{k/P}\times m_k\mathbf a_{k/P}$, since
$\mathbf v_{k/P}\times\mathbf v_{k/P}=\mathbf0$. The kinematic accelerations
are themselves obtained by symbolic differentiation.

Writing $S_1=m_1+m_2+m_3$, $S_2=m_2+m_3$, $S_3=m_3$, and
$\Delta_{ij}=\theta_i-\theta_j$, the resulting matrix can be presented
compactly as
\begin{equation}
    \boxed{
    \mathbf M(\boldsymbol\theta)=
    \begin{pmatrix}
        S_1l_1^2 & S_2l_1l_2\cos\Delta_{12}
                    & S_3l_1l_3\cos\Delta_{13}\\
        S_2l_1l_2\cos\Delta_{12} & S_2l_2^2
                    & S_3l_2l_3\cos\Delta_{23}\\
        S_3l_1l_3\cos\Delta_{13} & S_3l_2l_3\cos\Delta_{23}
                    & S_3l_3^2
    \end{pmatrix}
    }
    \label{eq:triple_ne_mass_matrix}
\end{equation}
with forcing components
\begin{align}
    b_1 &= -S_1gl_1\sin\theta_1
           -S_2l_1l_2\dot\theta_2^2\sin\Delta_{12}
           -S_3l_1l_3\dot\theta_3^2\sin\Delta_{13}, \\
    b_2 &= -S_2gl_2\sin\theta_2
           +S_2l_1l_2\dot\theta_1^2\sin\Delta_{12}
           -S_3l_2l_3\dot\theta_3^2\sin\Delta_{23}, \\
    b_3 &= -S_3gl_3\sin\theta_3
           +S_3l_1l_3\dot\theta_1^2\sin\Delta_{13}
           +S_3l_2l_3\dot\theta_2^2\sin\Delta_{23}.
    \label{eq:triple_ne_forcing}
\end{align}
The three coupled equations of motion are therefore
\begin{equation}
    \boxed{\mathbf M(\boldsymbol\theta)\ddot{\boldsymbol\theta}
           =\mathbf b(\boldsymbol\theta,\dot{\boldsymbol\theta}).}
    \label{eq:triple_ne_final}
\end{equation}
For strictly positive point masses and rod lengths, $\mathbf M$ is positive
definite at every configuration, so the angular accelerations are uniquely
determined. Indeed, for any nonzero vector of angular velocities,
$\dot{\boldsymbol\theta}^{\top}\mathbf M\dot{\boldsymbol\theta}
=\sum_{k=1}^{3}m_k\|\mathbf v_k\|^2>0$, because the triangular
kinematic relations imply that not all $\mathbf v_k$ can vanish unless
all angular velocities vanish. The state vector is
$\mathbf u=(\theta_1,\theta_2,\theta_3,
\dot\theta_1,\dot\theta_2,\dot\theta_3)^\top$; at each solver evaluation,
\texttt{system.jl} constructs $\mathbf M$ and $\mathbf b$, solves the linear
system using Julia's backslash operator, and returns the first-order time
derivatives. The resulting ODE is integrated using the \texttt{Tsit5} method
from \texttt{DifferentialEquations.jl}, with absolute and relative tolerances
of $10^{-8}$. The symbolic implementation also evaluates the resulting
matrix and forcing vector numerically at a test configuration and checks them
against the compact expressions used by the numerical solver.

The equations coincide with the equations obtained using the
Euler--Lagrange formalism. Their agreement provides an independent check of
the dynamics, while the explicit moving-pivot terms show why the rotational
moment equation for a fixed pivot cannot be applied unmodified at joints $A$
and $B$. Trajectory, angular-displacement, and energy-conservation plots, as
well as a pendulum animation, are generated separately by
\texttt{animations.jl} and called from \texttt{main.jl}.

The results from the symbolic computer are massive and not human readable.

   moving-pivot correction = (cos(θ2)*(cos(θ1)*l1*α1 - l1*sin(θ1)*(ω1^2))*l2 + (l1*sin(θ1)*α1 + cos(θ1)*l1*(ω1^2))*l2*sin(θ2))*m2 + ((l1*sin(θ1)*α1 + cos(θ1)*l1*(ω1^2))*(l2*sin(θ2) + l3*sin(θ3)) - (-cos(θ2)*l2 - cos(θ3)*l3)*(cos(θ1)*l1*α1 - l1*sin(θ1)*(ω1^2)))*m3
   gravitational torque = -g*l2*m2*sin(θ2) - g*(l2*sin(θ2) + l3*sin(θ3))*m3

   H_rel = ((cos(θ3)^2)*(l3^2)*ω3 + (l3^2)*(sin(θ3)^2)*ω3)*m3
   moving-pivot correction = (cos(θ3)*(cos(θ1)*l1*α1 + cos(θ2)*l2*α2 - l1*sin(θ1)*(ω1^2) - l2*sin(θ2)*(ω2^2))*l3 + (l1*sin(θ1)*α1 + l2*sin(θ2)*α2 + cos(θ1)*l1*(ω1^2) + cos(θ2)*l2*(ω2^2))*l3*sin(θ3))*m3
